name: HRM migration metadata preflight

# Dedicated read-only preparation. PRs exercise only isolated source/SQL tests.
# Production access is explicitly dispatched from the exact protected main head.
on:
  pull_request:
    paths:
      - .github/workflows/hrm-migration-metadata-preflight.yml
      - scripts/hrm-migration-metadata-preflight.mjs
      - scripts/hrm-migration-metadata-preflight.sql
      - scripts/hrm-migration-metadata-preflight.test.mjs
      - docs/hrm-release-preflight-session-log.md
      - prisma/migrations/20260829114500_workforce_future_only_lifecycle/migration.sql
      - prisma/migrations/20260927014000_workforce_exception_case_revisions/migration.sql
      - prisma/migrations/20260927014100_workforce_exception_case_revisions_backfill/migration.sql
  workflow_dispatch:
    inputs:
      expected_main_sha:
        description: Exact current main and deployed artifact SHA (40 lowercase characters)
        required: true
        type: string

permissions:
  contents: read
  actions: read

jobs:
  source-validation:
    if: ${{ github.event_name == 'pull_request' || github.event_name == 'workflow_dispatch' }}
    runs-on: ubuntu-24.04
    timeout-minutes: 5
    concurrency:
      group: hrm-metadata-source-${{ github.event.pull_request.number || github.ref }}
      cancel-in-progress: true
    services:
      postgres:
        image: postgres:16
        env:
          POSTGRES_DB: hrm_preflight_test
          POSTGRES_USER: postgres
          POSTGRES_PASSWORD: isolated-preflight-only
        ports:
          - 5432/tcp
        options: >-
          --health-cmd "pg_isready -U postgres -d hrm_preflight_test"
          --health-interval 5s --health-timeout 5s --health-retries 10
    steps:
      - uses: actions/checkout@v6
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
      - name: Reject dispatch outside exact main before any inspection
        if: ${{ github.event_name == 'workflow_dispatch' }}
        env:
          EXPECTED_MAIN_SHA: ${{ inputs.expected_main_sha }}
        run: |
          set -euo pipefail
          [ "$GITHUB_REF" = refs/heads/main ] \
            && [[ "$EXPECTED_MAIN_SHA" =~ ^[0-9a-f]{40}$ ]] \
            && [ "$EXPECTED_MAIN_SHA" = "$GITHUB_SHA" ] || {
              echo '::error::Metadata dispatch requires an exact main revision'
              exit 1
            }
      - name: Check bounded Node sources and isolated SQL contracts
        env:
          CI: 'true'
          HRM_PREFLIGHT_TEST_DATABASE_URL: postgresql://postgres:isolated-preflight-only@127.0.0.1:${{ job.services.postgres.ports[5432] }}/hrm_preflight_test
        run: |
          set -euo pipefail
          node --check scripts/hrm-migration-metadata-preflight.mjs
          node --check scripts/hrm-migration-metadata-preflight.test.mjs
          node --test scripts/hrm-migration-metadata-preflight.test.mjs
          node scripts/ci/check-github-runner-policy.mjs
          git diff --check

  production-metadata:
    needs: source-validation
    if: ${{ github.event_name == 'workflow_dispatch' && github.ref == 'refs/heads/main' }}
    runs-on: ubuntu-24.04
    timeout-minutes: 5
    environment: production
    concurrency:
      group: production-deploy
      cancel-in-progress: false
    steps:
      - name: Checkout exact dispatch revision
        uses: actions/checkout@v6
        with:
          fetch-depth: 1
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
      - name: Require protected production environment
        uses: ./.github/actions/assert-production-environment-protection
        with:
          github-token: ${{ github.token }}
      - name: Bind inspection to exact current main
        env:
          EXPECTED_MAIN_SHA: ${{ inputs.expected_main_sha }}
        run: |
          set -euo pipefail
          [ "$GITHUB_REF" = refs/heads/main ] || exit 1
          [[ "$EXPECTED_MAIN_SHA" =~ ^[0-9a-f]{40}$ ]] || {
            echo '::error::expected_main_sha must be an exact lowercase Git SHA'
            exit 1
          }
          git fetch --no-tags --depth=1 origin refs/heads/main
          [ "$GITHUB_SHA" = "$(git rev-parse FETCH_HEAD)" ] \
            && [ "$EXPECTED_MAIN_SHA" = "$GITHUB_SHA" ] || {
              echo '::error::Inspection requires the exact current main head'
              exit 1
            }
      - name: Setup existing pinned production SSH
        uses: ./.github/actions/setup-production-ssh
        with:
          ssh-private-key: ${{ secrets.SSH_PRIVATE_KEY }}
          server-host: ${{ secrets.SERVER_HOST }}
          known-hosts: ${{ secrets.SERVER_SSH_KNOWN_HOSTS }}
      - name: Inspect fixed metadata in memory and retain only validated output
        env:
          EXPECTED_MAIN_SHA: ${{ inputs.expected_main_sha }}
          PRODUCTION_SSH_USER: ${{ secrets.SERVER_USER }}
          PRODUCTION_SSH_HOST: ${{ secrets.SERVER_HOST }}
        run: |
          set -euo pipefail
          umask 077
          set +e
          DIAGNOSTIC_OUTPUT="$(
            node scripts/hrm-migration-metadata-preflight.mjs --emit-remote \
              | timeout 70s ssh -T -o ConnectTimeout=10 -o LogLevel=ERROR -i ~/.ssh/deploy_key \
                  "$PRODUCTION_SSH_USER@$PRODUCTION_SSH_HOST" \
                  "EXPECTED_MAIN_SHA='$EXPECTED_MAIN_SHA' NODE_NO_WARNINGS=1 node --input-type=module -" 2>/dev/null
          )"
          DIAGNOSTIC_STATUS=$?
          set -e
          # The URL, environment file, raw query output and stderr never reach
          # runner files or logs. Reject unknown fields before writing evidence.
          if ! printf '%s' "$DIAGNOSTIC_OUTPUT" \
            | node scripts/hrm-migration-metadata-preflight.mjs --validate-output "$EXPECTED_MAIN_SHA" \
                > "$RUNNER_TEMP/hrm-metadata-result.json"; then
            printf '{"version":1,"status":"ERROR","code":"TRANSPORT_OR_OUTPUT_INVALID","expectedMainSha":"%s"}\n' \
              "$EXPECTED_MAIN_SHA" > "$RUNNER_TEMP/hrm-metadata-result.json"
            echo '::error::Read-only metadata transport or sanitized output validation failed'
            exit 1
          fi
          [ "$DIAGNOSTIC_STATUS" -eq 0 ] || {
            echo '::error::Metadata inspection is incomplete or failed; sanitized original result retained'
            exit 1
          }
          git fetch --no-tags --depth=1 origin refs/heads/main
          [ "$EXPECTED_MAIN_SHA" = "$(git rev-parse FETCH_HEAD)" ] || {
            echo '::error::Main changed during inspection; repeat at the new exact deployed revision'
            exit 1
          }
      - name: Preserve sanitized result including original incomplete/error outcomes
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: hrm-migration-metadata-${{ github.run_id }}-${{ github.run_attempt }}
          path: ${{ runner.temp }}/hrm-metadata-result.json
          if-no-files-found: warn
          retention-days: 30
      - name: Remove ephemeral runner SSH material
        if: always()
        run: rm -f ~/.ssh/deploy_key ~/.ssh/known_hosts ~/.ssh/config
